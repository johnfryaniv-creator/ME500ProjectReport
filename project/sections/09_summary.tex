\section{Summary}

\begin{enumerate}
  \item The Kelvin--Voigt rod obeys
        $u_{tt} = c^2 u_{xx} + \nu\, u_{xxt}$, with $c^2 = E/\rho$ and $\nu = \eta/\rho$.
  \item The ansatz $u = \Real\{X(x)e^{i\omega t}\}$ reduces this to
        $X'' + k^2 X = 0$ with complex wavenumber
        $k^2 = \omega^2/\big[c^2(1+i\omega\tau)\big]$, where $\tau = \eta/E$.
  \item Imposing $X(0)=U_0$ and $X(L)=0$ gives
        $X = U_0\sin\!\big(k(L-x)\big)/\sin(kL)$.
  \item In real form,
        $u = U_0\,\mathcal{A}(x)\cos\!\big(\omega t + \theta(x)\big)$, with
        $\mathcal{A}$ and $\theta$ given by \cref{eq:amplitude,eq:phase} and
        $k = \beta - i\alpha$ given by \cref{eq:alpha_beta}.
\end{enumerate}
